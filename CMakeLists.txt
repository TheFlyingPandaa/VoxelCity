cmake_minimum_required(VERSION 3.24)
project(VoxelCity VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "VoxelCity requires Windows and DirectX 12")
endif()
include(FetchContent)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.9b GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(imgui)
add_library(imgui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp src/UiRenderer.cpp)
target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_compile_definitions(imgui PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(imgui PUBLIC d3d12 dxgi d3dcompiler dwmapi)
add_library(voxel_world src/World.cpp src/Voxels.cpp src/Vegetation.cpp src/RailWorld.cpp src/RoadNetwork.cpp)
target_include_directories(voxel_world PUBLIC src)
target_compile_definitions(voxel_world PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
add_library(voxel_traffic src/Traffic.cpp src/NetworkWorker.cpp)
target_link_libraries(voxel_traffic PUBLIC voxel_world)
target_compile_options(voxel_traffic PRIVATE /W4 /permissive- /EHsc /arch:AVX2)
add_library(voxel_city src/City.cpp src/Rail.cpp src/RailCity.cpp)
target_link_libraries(voxel_city PUBLIC voxel_traffic)
target_compile_options(voxel_city PRIVATE /W4 /permissive- /EHsc)
add_executable(VoxelCity WIN32 src/Main.cpp src/Renderer.cpp src/VoxelRenderer.cpp src/RenderWorker.cpp)
target_link_libraries(VoxelCity PRIVATE voxel_world voxel_traffic voxel_city imgui d3d12 dxgi dxguid comdlg32 shell32)
if(MSVC)
  target_compile_options(voxel_world PRIVATE /W4 /permissive- /EHsc)
  target_compile_options(VoxelCity PRIVATE /W4 /permissive- /EHsc)
endif()
file(GLOB SDK_DXC "C:/Program Files (x86)/Windows Kits/10/bin/*/x64/dxc.exe")
list(SORT SDK_DXC COMPARE NATURAL ORDER DESCENDING)
find_program(DXC_EXECUTABLE dxc HINTS "$ENV{WindowsSdkVerBinPath}/x64")
if(NOT DXC_EXECUTABLE AND SDK_DXC)
  list(GET SDK_DXC 0 DXC_EXECUTABLE)
endif()
if(NOT DXC_EXECUTABLE)
  message(FATAL_ERROR "Install a Windows SDK containing DXC (10.0.22621 or newer).")
endif()
set(SHADER_OUTPUTS)
foreach(NAME IN ITEMS World Cars Buildings)
string(TOLOWER ${NAME} LOWER_NAME)
foreach(STAGE IN ITEMS vs ps)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/${LOWER_NAME}.${STAGE}.cso
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders
    COMMAND "${DXC_EXECUTABLE}" -T ${STAGE}_6_5 -E ${STAGE}Main -O3 -Ges
      -Fo ${CMAKE_CURRENT_BINARY_DIR}/shaders/${LOWER_NAME}.${STAGE}.cso ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${NAME}.hlsl
    DEPENDS shaders/${NAME}.hlsl shaders/World.hlsl VERBATIM)
list(APPEND SHADER_OUTPUTS ${CMAKE_CURRENT_BINARY_DIR}/shaders/${LOWER_NAME}.${STAGE}.cso)
endforeach()
endforeach()
foreach(STAGE IN ITEMS vs surface light resolve post test)
  if(STAGE STREQUAL "vs")
    set(VOXEL_PROFILE vs_6_5)
    set(VOXEL_ENTRY vsMain)
  elseif(STAGE STREQUAL "test")
    set(VOXEL_PROFILE cs_6_5)
    set(VOXEL_ENTRY testMain)
  else()
    set(VOXEL_PROFILE ps_6_5)
    set(VOXEL_ENTRY ${STAGE}Main)
  endif()
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/shaders/voxel.${STAGE}.cso
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders
    COMMAND "${DXC_EXECUTABLE}" -T ${VOXEL_PROFILE} -E ${VOXEL_ENTRY} -O3 -Ges
      -Fo ${CMAKE_CURRENT_BINARY_DIR}/shaders/voxel.${STAGE}.cso ${CMAKE_CURRENT_SOURCE_DIR}/shaders/Voxel.hlsl
    DEPENDS shaders/Voxel.hlsl VERBATIM)
  list(APPEND SHADER_OUTPUTS ${CMAKE_CURRENT_BINARY_DIR}/shaders/voxel.${STAGE}.cso)
endforeach()
add_custom_target(shaders DEPENDS ${SHADER_OUTPUTS})
add_dependencies(VoxelCity shaders)
add_custom_command(TARGET VoxelCity POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_BINARY_DIR}/shaders $<TARGET_FILE_DIR:VoxelCity>/shaders
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/data $<TARGET_FILE_DIR:VoxelCity>/data
  COMMAND ${CMAKE_COMMAND} -E copy_if_different ${imgui_SOURCE_DIR}/LICENSE.txt $<TARGET_FILE_DIR:VoxelCity>/DearImGui-LICENSE.txt)
enable_testing()

target_link_libraries(WorldTests PRIVATE voxel_world)
add_test(NAME WorldTests COMMAND WorldTests)

add_executable(TrafficTests tests/TrafficTests.cpp)
target_link_libraries(TrafficTests PRIVATE voxel_traffic)
add_test(NAME TrafficTests COMMAND TrafficTests)
add_executable(CityTests tests/CityTests.cpp)
target_link_libraries(CityTests PRIVATE voxel_city)
add_test(NAME CityTests COMMAND CityTests)
add_executable(CityBenchmark tests/CityBenchmark.cpp)
target_link_libraries(CityBenchmark PRIVATE voxel_city psapi)
add_executable(VegetationTests tests/VegetationTests.cpp)
target_link_libraries(VegetationTests PRIVATE voxel_city)
add_test(NAME VegetationTests COMMAND VegetationTests)
add_executable(HighwayTests tests/HighwayTests.cpp)
target_link_libraries(HighwayTests PRIVATE voxel_city)
add_test(NAME HighwayTests COMMAND HighwayTests)

add_executable(VoxelTests tests/VoxelTests.cpp)
target_link_libraries(VoxelTests PRIVATE voxel_world)
add_test(NAME VoxelTests COMMAND VoxelTests)

target_link_libraries(VoxelGpuTests PRIVATE voxel_world d3d12 dxgi dxguid d3dcompiler)
